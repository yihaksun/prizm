2. 파라미터 기반 모델 학습

입력된 변수만 바꾸면 같은 Notebook으로 다른 데이터와 기준 모델을 재사용할 수 있습니다.

In [4]:
# PRIZM은 인증, Tracking URI, Experiment, 공통 lineage tag만 설정합니다.
tracking = prizm.tracking.configure_mlflow(
    project_id="PRJ000212"
)
mlflow.set_experiment(tracking.experiment_name)
run = mlflow.start_run(
    run_name=f"weld-yolo12-{data_id}",
    log_system_metrics=True,
)
mlflow.log_params({
    "data_id": data_id,
    "base_model_id": model_id,
    "base_model_version": model_version,
    "epochs": epochs,
    "batch_size": batch_size,
    "image_size": image_size,
    "seed": seed,
})

model = YOLO(base_model.path)
result = model.train(
    data=dataset.manifest,
    epochs=epochs,
    batch=batch_size,
    imgsz=image_size,
    seed=seed,
)
mlflow.log_artifact(result.best, artifact_path="model")
